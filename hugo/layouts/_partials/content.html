{{- /* A page's content, with the images it links at full size swapped for processed copies: the 4.2 MB hero photo, and every video poster it references. The poster swap used to be a hardcoded `replace` of exactly one filename, which worked for the single recording on the site and silently did nothing for any other poster path. It now walks the content for every poster="/images/" reference and resolves each one against assets/images/, so adding a recording means adding its poster and pointing at it. No partial change required. */ -}} {{- $a := partialCached "assets.html" . "assets" -}} {{- $c := replace .Content `src="/grahambrooks.png"` (printf `src="%s" width="64" height="64"` $a.avatar.RelPermalink) -}} {{- range findRESubmatch `poster="/images/([^"]+)"` $c -}} {{- $name := index . 1 -}} {{- with resources.Get (printf "images/%s" $name) -}} {{- $c = replace $c (printf `poster="/images/%s"` $name) (printf `poster="%s"` (.Resize "800x webp q75").RelPermalink) -}} {{- else -}} {{- warnf "content.html: no such poster in assets/images: %s" $name -}} {{- end -}} {{- end -}} {{- $c | safeHTML -}}
